\begin{problem}[39.2]
  Show that given $\alpha = \sqrt{2} + \sqrt{3} \in \C$ is algebraic over $\Q$ by finding $f(x) \in \Q[x]$ such that $f(\alpha) = 0$.
\end{problem}

\begin{solution}
  Squaring $\alpha$ gives $\alpha^2 = 5 + 2\sqrt{6}$. Squaring again gives $\alpha^4 = 49 + 20\sqrt{6}$. Thus, we have $\alpha^4 - 10\alpha^2 + 1 = 0$, so $f(x) = x^4 - 10x^2 + 1$ is a polynomial in $\Q[x]$ such that $f(\alpha) = 0$.
\end{solution}

\begin{problem}[39.10]
  Classify $\alpha = 1 + i$ as algebraic or transcendental over the field $F = \R$. If $\alpha$ is algebraic over $F$, find $\deg(\alpha, F)$.
\end{problem}

\begin{solution}
  The number $\alpha$ is algebraic over $\R$, since it is a zero of a polynomial, $f(x) \in \R[x]$. To find this polynomial, we square $\alpha$ to get $\alpha^2 = 2i$. Squaring again gives $\alpha^4 = -4$. Thus, we have $\alpha^4 + 4 = 0$, so $f(x) = x^4 + 4$ is a polynomial in $\R[x]$ such that $f(\alpha) = 0$. Notice that $f(x)$ is reducible over $\R$, specifically, we have:
  \[%
    f(x) = \left(x^2 + 2x + 2\right)\left(x^2 - 2x + 2\right) = p(x)q(x)
  .\]%
  Notice that $\alpha$ is a root of $q(x)$, so $\irr(\alpha, \R) = q(x) = x^2 - 2x + 2$ and $\deg(\alpha, \R) = 2$.
\end{solution}

\begin{problem}[39.16]
  Classify $\alpha = \pi^2$ as algebraic or transcendental over the field $F = \Q(\pi^3)$. If $\alpha$ is algebraic over $F$, find $\deg(\alpha, F)$.
\end{problem}

\begin{solution}
  We first find what the simple field extension $\Q(\pi^3)$ looks like. Notice that $\Q(\pi^3) = \{a + b\pi^3 : a, b \in \Q\}$, since $\pi^3$ is transcendental over $\Q$. Assume that $\alpha$ is algebraic over $F$. Then, there exists a nonzero polynomial $f(x) \in F[x]$ such that $f(\alpha) = 0$. We can write $f(x)$ as $f(x) = a_nx^n + \cdots a_1x^1 + a_0$. Since $a_i \in F$ for all $i$, we can write $a_i = c_i + d_i\pi^3$ for some $c_i, d_i \in \Q$. Thus, we have:
  \[%
    f(x) = (c_n + d_n\pi^3)x^n + \cdots + (c_1 + d_1\pi^3)x^1 + (c_0 + d_0\pi^3)
  .\]%
  We can rearrange this to get:
  \[%
    f(x) = (c_nx^n + \cdots + c_1x^1 + c_0) + (d_nx^n + \cdots + d_1x^1 + d_0)\pi^3
  .\]%
  Since $f(\alpha) = 0$, we have:
  \[%
    (c_n\alpha^n + \cdots + c_1\alpha^1 + c_0) + (d_n\alpha^n + \cdots + d_1\alpha^1 + d_0)\pi^3 = 0
  .\]%
  This implies that $d_n\alpha^n + \cdots + d_1\alpha^1 + d_0 = 0$, since $\pi^3$ is transcendental over $\Q$. Thus, we have $c_n\alpha^n + \cdots + c_1\alpha^1 + c_0 = 0$. This implies that $\alpha$ is algebraic over $\Q$, which is a contradiction. Thus, $\alpha$ is transcendental over $F$.
\end{solution}

\begin{problem}[39.24]
  We have stated without proof that $\pi$ and $e$ are transcendental over $\Q$.
  \begin{enumerate}
    \item Find a subfield $F$ of $\R$ such that $\pi$ is algebraic of degree 3 over $F$.
    \item Find a subfield $E$ of $\R$ such that $e^2$ is algebraic of degree 5 over $E$.
  \end{enumerate}
\end{problem}

\begin{solution}[(i)]
  We'll construct the subfield $F$ of $\R$ as follows. Let $F = \{a + b\pi + c\pi^2 + d\pi^3 : a, b, c, d \in \Q\}$. You can easily check that this is indeed a subfield of $\R$. Next, we show that $\pi$ is algebraic of degree 3 over $F$. Notice that $\pi$ is a root of the polynomial $f(x) = x^3 - \pi^3 \in F[x]$. To show that $\pi$ is algebraic of degree 3 over $F$, we need to show that $f(x)$ is irreducible over $F$. Assume that $f(x)$ is reducible over $F$. Then, there exists a non constant polynomial $g(x) \in F[x]$ such that $g(x) \mid f(x)$. Since $f(x)$ is a cubic polynomial, we must have $\deg(g(x)) = 1$. Thus, we can write $g(x) = x - r$ for some $r \in F$. Since $g(x) \mid f(x)$, we have $f(r) = 0$, which implies that $r^3 = \pi^3$. This implies that $r = \pi$, since $\pi$ is positive. However, this contradicts the fact that $r \in F$, since $\pi$ is transcendental over $\Q$. Thus, $f(x)$ is irreducible over $F$, so $\pi$ is algebraic of degree 3 over $F$.
\end{solution}

\begin{solution}[(ii)]
  We can construct the subfield $E$ of $\R$ similarly to how we did in part (i). Take $E = \{a + \cdots ge^5 : a, b, c, d, g \in \Q\}$. Again, you can easily check that this is indeed a subfield of $\R$. Next, we show that $e^2$ is algebraic of degree 5 over $E$. Notice that $e^2$ is a root of the polynomial $f(x) = x^5 - e^{10} \in E[x]$. To show that $e^2$ is algebraic of degree 5 over $E$, we need to show that $f(x)$ is irreducible over $E$. Assume that $f(x)$ is reducible over $E$. Then, there exists a non constant polynomial $g(x) \in E[x]$ such that $g(x) \mid f(x)$. Since $f(x)$ is a quintic polynomial, we must have $\deg(g(x)) = 1$ or $\deg(g(x)) = 2$. If $\deg(g(x)) = 1$, then we can write $g(x) = x - r$ for some $r \in E$. Since $g(x) \mid f(x)$, we have $f(r) = 0$, which implies that $r^5 = e^{10}$. This implies that $r = e^2$, since $e$ is positive. However, this contradicts the fact that $r \in E$, since $e$ is transcendental over $\Q$. If $\deg(g(x)) = 2$, then we can write $g(x) = x^2 + ax + b$ for some $a, b \in E$. Since $g(x) \mid f(x)$, we have:
  \[%
    f(x) = (x^2 + ax + b)(x^3 + cx^2 + dx + e)
  .\]%
  Expanding the right hand side gives us:
  \[%
    f(x) = x^5 + (a + c)x^4 + (b + ac + d)x^3 + (bc + ad + e)x^2 + (bd + ae)x + be
  .\]%
  Comparing coefficients with the left hand side gives us the following system of equations:
  \begin{align*}
    a + c &= 0 \\
    b + ac + d &= 0 \\
    bc + ad + e &= 0 \\
    bd + ae &= 0 \\
    be &= -e^{10}
  .\end{align*}
  From the first equation, we have $c = -a$. Substituting this into the second equation gives us $b - a^2 + d = 0$, which implies that $d = a^2 - b$. Substituting $c = -a$ and $d = a^2 - b$ into the third equation gives us $-ab + a^3 - be + e = 0$, which implies that $e = ab - a^3 + be$. Substituting $c = -a$, $d = a^2 - b$, and $e = ab - a^3 + be$ into the fourth equation gives us $-b^2 + a^4 - abe + a^3b - b^2e + abe = 0$, which simplifies to $-b^2 + a^4 + a^3b - b^2e = 0$. Finally, substituting $c = -a$, $d = a^2 - b$, and $e = ab - a^3 + be$ into the fifth equation gives us $-b(ab - a^3 + be) = -e^{10}$, which simplifies to $-ab^2 + a^4b - b^3e = -e^{10}$. Thus, we have the following system of equations:
  \begin{align*}
    c &= -a \\
    d &= a^2 - b \\
    e &= ab - a^3 + be \\
    -b^2 + a^4 + a^3b - b^2e &= 0 \\
    -ab^2 + a^4b - b^3e &= -e^{10}
  .\end{align*}
  From the fourth equation, we have $b^2e = -b^2 + a^4 + a^3b$. Substituting this into the fifth equation gives us $-ab^2 + a^4b - b^3(-b^2 + a^4 + a^3b) = -e^{10}$, which simplifies to $-ab^2 + a^4b + b^5 - a^4b^3 - a^3b^4 = -e^{10}$. Rearranging gives us $-ab^2 + a^4b + b^5 - a^4b^3 - a^3b^4 + e^{10} = 0$. This is a polynomial in $a$ and $b$ with coefficients in $E$. Since $E$ is an extension of $\Q$, we can view this as a polynomial in $a$ and $b$ with coefficients in $\Q$. Since $\Q$ is infinite, there exist infinitely many pairs $(a, b) \in \Q \times \Q$ such that this polynomial evaluates to zero. Thus, there exist infinitely many pairs $(a, b) \in E \times E$ such that this polynomial evaluates to zero. This contradicts the fact that there are only finitely many pairs $(a, b) \in E \times E$. Thus, $f(x)$ is irreducible over $E$, so $e^2$ is algebraic of degree 5 over $E$.
\end{solution}

\begin{problem}[39.29]
  Let $E$ be an extension field of $F$, and let $\alpha, \beta \in E$. Suppose $\alpha$ is transcendental over $F$ but algebraic over $F(\beta)$. Show that $\beta$ is algebraic over $F(\alpha)$.
\end{problem}

\begin{solution}
  Let $f(x) = \irr(\alpha, F(\beta)) = a_nx^n + \cdots a_1x^1 + a_0$. Notice that $f(\alpha) = 0$, so we have $a_n\alpha^n + \cdots + a_1\alpha^1 + a_0 = 0$. Rearranging gives us $a_n\alpha^n + \cdots + a_1\alpha^1 = -a_0$. Since $\alpha$ is transcendental over $F$, we can view this as a polynomial in $\beta$ with coefficients in $F(\alpha)$. Thus, we have $g(\beta) = a_n\alpha^n + \cdots + a_1\alpha^1 + a_0 = 0$, where $g(x) \in F(\alpha)[x]$. This implies that $\beta$ is algebraic over $F(\alpha)$.
\end{solution}

\begin{problem}[39.30]
  Let $E$ be an extension field of a finite field $F$, where $F$ is $q$ elements. Let $\alpha \in E$ be algebraic over $F$ of degree $n$. Prove that $F(\alpha)$ has $q^n$ elements.
\end{problem}

\begin{solution}
  Let $\irr(\alpha, F) = a_nx^n + \cdots + a_1x^1 + a_0$. Notice that $F(\alpha) = \{a_{n - 1}\alpha^{n - 1} + \cdots + a_1\alpha^1 + a_0 : a_i \in F\}$, since $\irr(\alpha, F)$ is irreducible over $F$. Since $F$ has $q$ elements, there are $q$ choices for each of the coefficients $a_i$. Thus, there are $q^n$ elements in $F(\alpha)$.
\end{solution}
